function getNDBI(image,nirBand,swirBand){
  return image.normalizedDifference([swirBand,nirBand]).rename("NDBI")
}

function maskClouds(image){
 var qa = image.select("QA60") ;
 var cloudBitMask = 1 << 10;
  var cirrusBitMask = 1 << 11;

  // Both flags should be set to zero, indicating clear conditions.
  var mask = qa.bitwiseAnd(cloudBitMask).eq(0)
      .and(qa.bitwiseAnd(cirrusBitMask).eq(0));

  return image.updateMask(mask).divide(10000); // Scaling  the bands 
  
}

// 3. Import the wards
var wards = ee.FeatureCollection("projects/researchgee/assets/PokharaMap");

// NEW STEP: Extract the last 2 digits from 'DDGNWW' and create a 'Ward_No' column
var wardsProcessed = wards.map(function(feature) {
  // Get the DDGNWW value as an Earth Engine Number
  var ddgnwwValue = ee.Number(feature.get('DDGNWW'));
  
  // Convert it to an integer (to remove any decimals)
  var intValue = ddgnwwValue.toInt();
  
  // Use modulo 100 (.mod) to get the remainder (which gives us the last 2 digits)
  var wardNumber = intValue.mod(100);
  
  // Set this new number as 'Ward_No' and return the updated feature
  return feature.set('Ward_No', wardNumber);
});

// 4. Calculate area per ward using the PROCESSED wards
var wardChangeStats = changeAreaImage.reduceRegions({
  collection: wardsProcessed, // Make sure to use the new 'wardsProcessed' variable here!
  reducer: ee.Reducer.sum(),
  scale: 10 // Sentinel-2 scale
});
